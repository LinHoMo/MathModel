\section{模型假设}
\item 总耕地面积 A 固定，可分配给多种作物
\item 各作物单位面积收益 r_i 已知或可估计
\item 作物间存在轮作约束（同一种作物不能连续种植）
\item 市场需求在计划期内稳定
\item 考虑收益最大化和风险最小化两个目标

\section{模型建立与求解}
\subsection{收益模型}
\begin{equation}
R = Σ r_i · x_i
\end{equation}
其中，线性收益。

\subsection{面积约束}
\begin{equation}
Σ x_i ≤ A
\end{equation}
其中，耕地约束。

\subsection{轮作约束}
\begin{equation}
x_i(t) + x_i(t+1) ≤ A（简化）
\end{equation}
其中，轮作约束。

\subsection{风险模型}
\begin{equation}
Var(R) = Σ σ_i²·x_i² + 2·Σ_{i<j} ρ_ij·σ_i·σ_j·x_i·x_j
\end{equation}
其中，方差-协方差。


\subsection{约束条件}
\item Σx_i ≤ A（耕地面积约束）
\item x_i ≥ 0（非负约束）

\subsection{目标函数}
\begin{equation}
max R = Σ r_i · x_i
\end{equation}
其中，总收益最大。
