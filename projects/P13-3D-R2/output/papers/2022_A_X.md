\section{模型假设}
\item 浮体做单自由度垂向振动（升沉运动）
\item 波浪激励为线性规则波（Airy 波理论）
\item 阻尼元件为线性粘性阻尼（PTO 系统）
\item 浮体完全浸没或部分浸没，附加质量为常数
\item 忽略波浪辐射阻尼和兴波阻力
\item 能量转换效率为常数（发电机效率）

\section{模型建立与求解}
\subsection{振动方程（牛顿第二定律）}
\begin{equation}
m·ẍ + c·ẋ + k·x = F0·sin(ωt)
\end{equation}
其中，m 含附加质量，c 为 PTO 阻尼。

\subsection{稳态振幅}
\begin{equation}
A = F0 / √((k - mω²)² + (cω)²)
\end{equation}
其中，受迫振动稳态解析解。

\subsection{输出功率}
\begin{equation}
P = η·c·v² = η·c·A²·ω²·cos²(ωt)
\end{equation}
其中，阻尼耗散功率×发电机效率。

\subsection{入射波功率}
\begin{equation}
P_wave = (ρ_w·g·A²·ω)/(2k_w) · S
\end{equation}
其中，Airy 波理论单位宽度功率。

\subsection{capture width ratio}
\begin{equation}
CWR = P_out / P_wave
\end{equation}
其中，无量纲效率指标。


\subsection{约束条件}
\item k > 0, c > 0（物理可实现）
\item |x| ≤ x_max（浮体行程限制）（机械限位）
\item 共振条件：ω_n = √(k/m) ≈ ω_wave（设计目标（Q3））

\subsection{目标函数}
\begin{equation}
max P_avg = η·c·A²·ω² / 2
\end{equation}
其中，Q3 优化目标：平均输出功率最大。

\begin{equation}
max capture_width_ratio = P_out / P_wave
\end{equation}
其中，能量捕获效率。


\section{模型评价与讨论}

\subsection{候选模型对比}
\item 单自由度受迫振动：优点 解析解可用，缺点 忽略非线性
\item 多自由度振动：优点 更精确，缺点 参数多

\subsection{选定模型}
选定模型：单自由度受迫振动

选择理由：波浪能装置可简化为单自由度振动系统

\subsection{不确定性分析}

\subsection{灵敏度分析}
\item 弹簧刚度 k：范围 0.5× 至 2× 共振值，指标 平均功率、振幅
\item PTO 阻尼 c：范围 0.5× 至 2× 最优值，指标 平均功率、CWR
\item 波浪频率 ω：范围 ±30% 设计频率，指标 功率频率响应曲线
\item 波浪幅值 F0：范围 0.5× 至 2×，指标 线性响应验证
