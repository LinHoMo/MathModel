\begin{abstract}
选定模型：ODE + 伯努利方程 题目要求分析压力变化规律，ODE 模型足够且可解释 \end{abstract}

\section{模型假设}
\begin{itemize}
\item 燃油近似不可压缩（体积弹性模量 E ≈ 常数）
\item 管内流动为准稳态（压力波传播时间 << 喷射周期）
\item 喷嘴流量遵循伯努利方程，流量系数 Cd 在工作范围内近似常数
\item 进油阀位移由凸轮型线决定，忽略阀弹簧动力学
\item 温度变化可忽略（短时喷射过程）
\end{itemize}

\section{模型建立}
\item 0 ≤ x_valve ≤ x_max
\item P ≥ 0（物理非负）
\item Q_in ≥ 0, Q_out ≥ 0
\item 一个喷射周期内 ∫Q_out dt = V_inject（喷油量）
\subsection{压力动态（体积弹性）}
\begin{equation}
dP/dt = (E/V)·(Q_in - Q_out)
\end{equation}

\subsection{进油流量}
\begin{equation}
Q_in = Cd·A_in(x_valve)·√(2(P_in - P)/ρ)
\end{equation}

\subsection{喷嘴流量}
\begin{equation}
Q_out = Cd·A_out·√(2P/ρ)
\end{equation}

\subsection{凸轮型线}
\begin{equation}
x_valve = f(cam_angle)
\end{equation}

\begin{equation}
min max|P(t) - P_target|（压力波动最小化）
\end{equation}

\begin{equation}
标定 Cd 使模型压力曲线与实测匹配
\end{equation}

\section{候选模型对比}
\begin{itemize}
\item ODE + 伯努利：优点 物理可解释，缺点 参数少
\item CFD 数值模拟：优点 精度高，缺点 计算成本高
\end{itemize}

选定模型：ODE + 伯努利方程

选择理由：题目要求分析压力变化规律，ODE 模型足够且可解释

\section{灵敏度分析}
\begin{itemize}
\item 流量系数 Cd：范围 ±20%，指标 压力波动幅值、峰值压力
\item 喷嘴面积 A_out：范围 ±15%，指标 压力响应、喷油速率
\end{itemize}
