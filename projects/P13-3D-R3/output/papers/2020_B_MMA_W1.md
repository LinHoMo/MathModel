\subsection{奖励函数}
\begin{equation}
R(s,a) = -cost(s,a) + reward(到达绿洲)
\end{equation}

\subsection{Bellman 方程}
\begin{equation}
V(s) = max_a [R(s,a) + γ·E[V(s')]
\end{equation}

\begin{equation}
min E[总时间] 或 max P(安全到达)
\end{equation}

\section{候选模型对比}
\begin{itemize}
\item 动态规划：优点 全局最优，缺点 状态空间大
\item MDP + 蒙特卡洛：优点 可处理连续状态，缺点 近似解
\end{itemize}

选定模型：动态规划 / MDP
