\begin{abstract}
选定模型：报童模型 + 价格优化 蔬菜单日决策适用报童框架 \end{abstract}

\section{模型假设}
\begin{itemize}
\item 蔬菜保鲜期为 T_max 天，超过则完全损耗
\item 需求受价格影响，线性近似 D = a - b·p
\item 每日补货一次，日末处理剩余库存
\item 历史销售数据可用于需求分布估计
\item 缺货损失 = 毛利率 × 期望缺货量
\end{itemize}

\section{模型建立}

\subsection{需求模型}
\begin{equation}
D_t = max(0, a - b·p_t + ε_t), ε_t ~ N(0, σ²)
\end{equation}

\subsection{库存动态}
\begin{equation}
w_t = max(w_{t-1} + q_t - D_t, 0)
\end{equation}

\subsection{损耗}
\begin{equation}
L_t = max(w_{t-1} + q_t - D_t - W_max, 0)
\end{equation}

\subsection{利润函数}
\begin{equation}
π_t = p_t·min(q_t + w_{t-1}, D_t) - c·q_t - h·L_t
\end{equation}

\begin{equation}
max E[日利润] = p·min(q,D) - c·q - 损耗成本
\end{equation}
