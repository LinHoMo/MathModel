\section{模型假设}
\begin{itemize}
\item 总耕地面积 A 固定，可分配给多种作物
\item 各作物单位面积收益 r_i 已知或可估计
\item 作物间存在轮作约束（同一种作物不能连续种植）
\item 市场需求在计划期内稳定
\item 考虑收益最大化和风险最小化两个目标
\end{itemize}

\section{模型建立}
\item Σx_i ≤ A
\item x_i ≥ 0
\subsection{收益模型}
\begin{equation}
R = Σ r_i · x_i
\end{equation}

\subsection{面积约束}
\begin{equation}
Σ x_i ≤ A
\end{equation}

\subsection{轮作约束}
\begin{equation}
x_i(t) + x_i(t+1) ≤ A（简化）
\end{equation}

\subsection{风险模型}
\begin{equation}
Var(R) = Σ σ_i²·x_i² + 2·Σ_{i<j} ρ_ij·σ_i·σ_j·x_i·x_j
\end{equation}

\begin{equation}
max R = Σ r_i · x_i
\end{equation}

\section{候选模型对比}
\begin{itemize}
\item 线性规划（选定）：优点 简单高效，缺点 忽略不确定性
\item 随机规划：优点 考虑价格波动，缺点 需要分布信息
\end{itemize}

选定模型：线性规划

选择理由：基础模型适用 LP
