\section{模型建立与求解}
\subsection{PCA 降维}
\begin{equation}
Z = X·W (W 为特征向量矩阵)
\end{equation}
其中，主成分分析。

\subsection{K-means 聚类}
\begin{equation}
argmin Σ||x_i - μ_k||²
\end{equation}
其中，欧氏距离聚类。


\subsection{约束条件}
\item 各成分含量 ≥ 0 且总和 = 100%（化学约束）

\subsection{目标函数}
\begin{equation}
最大化聚类/判别准确率
\end{equation}
其中，分类质量。
