\section{模型建立与求解}
\subsection{CLR 变换}
\begin{equation}
clr(x) = [ln(x_i/g(x))] 其中 g(x) 为几何均值
\end{equation}
其中，消除闭合约束。

\subsection{PCA}
\begin{equation}
Z = X·W, W = eigenvectors(Σ)
\end{equation}
其中，主成分分析。

\subsection{K-means}
\begin{equation}
迭代: 分配 x_i → nearest μ_k; 更新 μ_k = mean(cluster_k)
\end{equation}
其中，无监督聚类。

\subsection{LDA 判别}
\begin{equation}
w = S_w^{-1}(μ_1 - μ_2), score = w^T x
\end{equation}
其中，Fisher 线性判别。


\subsection{约束条件}
\item 各成分含量 ≥ 0 且总和 = 100%（化学约束）

\subsection{目标函数}
\begin{equation}
最大化聚类/判别准确率
\end{equation}
其中，分类质量。
