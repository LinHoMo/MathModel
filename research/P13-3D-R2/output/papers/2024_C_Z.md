\section{模型建立与求解}
\subsection{收益计算}
\begin{equation}
R = Σ r_i · x_i
\end{equation}
其中，线性收益。


\subsection{约束条件}
\item Σx_i ≤ A（耕地面积约束）
\item x_i ≥ 0（非负约束）

\subsection{目标函数}
\begin{equation}
max R = Σ r_i · x_i
\end{equation}
其中，总收益最大。
