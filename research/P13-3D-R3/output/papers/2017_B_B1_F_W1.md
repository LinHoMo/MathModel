\begin{abstract}
选定模型：Logistic 回归 定价问题需要可解释模型 \end{abstract}

\section{模型假设}
\begin{itemize}
\item 用户选择任务基于报酬-距离-难度的效用最大化
\item 任务完成率服从 logistic 函数（S 形曲线）
\item 区域固定效应可通过虚拟变量捕获
\item 时段效应（工作日/周末、白天/晚间）影响用户可用性
\item 任务难度与所需时间成正比
\end{itemize}

\section{模型建立}
\item 0 ≤ completion_rate ≤ 1
\item price ≥ 0
\item 总预算约束：Σ price_i ≤ B
\subsection{完成率模型（多因素 logistic）}
\begin{equation}
P(complete) = σ(β_p·price - β_d·distance - β_diff·difficulty + γ_r + δ_t + b)
\end{equation}

\subsection{收益函数}
\begin{equation}
Revenue = price · P(complete)
\end{equation}

\subsection{价格-完成率弹性}
\begin{equation}
ε = β_p · price · (1 - P(complete))
\end{equation}

\begin{equation}
max E[收益] = Σ price_i · completion_rate_i
\end{equation}

\begin{equation}
max 任务完成数 = Σ completion_rate_i
\end{equation}

\end{itemize}
