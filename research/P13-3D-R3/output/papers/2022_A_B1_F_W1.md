\begin{abstract}
选定模型：单自由度受迫振动 波浪能装置可简化为单自由度振动系统 \end{abstract}

\section{模型假设}
\begin{itemize}
\item 浮体做单自由度垂向振动（升沉运动）
\item 波浪激励为线性规则波（Airy 波理论）
\item 阻尼元件为线性粘性阻尼（PTO 系统）
\item 浮体完全浸没或部分浸没，附加质量为常数
\item 忽略波浪辐射阻尼和兴波阻力
\item 能量转换效率为常数（发电机效率）
\end{itemize}

\section{模型建立}
\item k > 0, c > 0
\item |x| ≤ x_max（浮体行程限制）
\item 共振条件：ω_n = √(k/m) ≈ ω_wave
\subsection{振动方程（牛顿第二定律）}
\begin{equation}
m·ẍ + c·ẋ + k·x = F0·sin(ωt)
\end{equation}

\subsection{稳态振幅}
\begin{equation}
A = F0 / √((k - mω²)² + (cω)²)
\end{equation}

\subsection{输出功率}
\begin{equation}
P = η·c·v² = η·c·A²·ω²·cos²(ωt)
\end{equation}

\subsection{入射波功率}
\begin{equation}
P_wave = (ρ_w·g·A²·ω)/(2k_w) · S
\end{equation}

\subsection{capture width ratio}
\begin{equation}
CWR = P_out / P_wave
\end{equation}

\begin{equation}
max P_avg = η·c·A²·ω² / 2
\end{equation}

\begin{equation}
max capture_width_ratio = P_out / P_wave
\end{equation}

\section{候选模型对比}
\begin{itemize}
\item 单自由度受迫振动（选定）：优点 解析解可用，缺点 忽略非线性
\item 多自由度振动：优点 更精确，缺点 参数多
\end{itemize}

选定模型：单自由度受迫振动

选择理由：波浪能装置可简化为单自由度振动系统

\section{灵敏度分析}
\begin{itemize}
\item 波浪频率 ω：范围 ±30% 设计频率，指标 功率频率响应曲线
\end{itemize}
