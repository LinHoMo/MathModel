\section{模型假设}
\begin{itemize}
\item 天气状态服从马尔可夫链（晴/阴/雨），转移概率已知
\item 资源消耗率与速度和天气相关
\item 绿洲位置和补给量已知
\item 队伍速度可连续调节
\item 目标为最小化期望到达时间或最大化安全概率
\end{itemize}

\section{模型建立}
\item r(t) ≥ 0 ∀t
\subsection{位置动态}
\begin{equation}
s(t+1) = s(t) + v(t)·Δt
\end{equation}

\subsection{资源消耗}
\begin{equation}
r(t+1) = r(t) - c(v, weather)·Δt
\end{equation}

\subsection{天气转移}
\begin{equation}
P(w'=j | w=i) = T_ij
\end{equation}

\subsection{Bellman 方程}
\begin{equation}
V(s,w) = min_a [cost(s,w,a) + γ·Σ_w' T(w,w')·V(s',w')]
\end{equation}

\begin{equation}
min E[总时间] 或 max P(安全到达)
\end{equation}

\section{候选模型对比}
\begin{itemize}
\item 动态规划：优点 全局最优，缺点 状态空间大
\item MDP + 蒙特卡洛：优点 可处理连续状态，缺点 近似解
\end{itemize}

选定模型：动态规划 / MDP
